\usepackage{graphicx}

%% True-to-size, left-justified figure whose caption width matches the image width.
%% Measures the image into a save-box, then wraps image and caption in a minipage of that
%% exact width. Use inside a figure/figure* environment:
%%   \tcfig{figures/foo.pdf}{Caption text.}{fig:label}
\newsavebox{\tcfigbox}
\newcommand{\tcfig}[3]{%
  \sbox{\tcfigbox}{\includegraphics{#1}}%
  \noindent\begin{minipage}{\wd\tcfigbox}%
  \usebox{\tcfigbox}\par
  \caption{#2}\label{#3}%
  \end{minipage}%
}

%% A multi-panel figure whose panels are separate true-size PDFs, laid out here rather
%% than composed in draw.io.
%%
%% WHY BOTH ROUTES EXIST. Every panel is written by a plot script as a true-size SVG, and
%% the same SVGs are embedded in the draw.io sources under notes/assets/drawio/. draw.io is
%% where the arrangement is workshopped by hand; this macro is what lets the document
%% compile from the panels alone in the meantime, at the same true size and with the same
%% 8 pt bold lowercase letters. Once a figure's arrangement is settled in draw.io, replace
%% the \panelrow block with a single \tcfig on the exported PDF.
%%
%% \panel{a}{figures/x.pdf} places one lettered panel; \panelrow{...} sets a row of them.
\newcommand{\panel}[2]{%
  \begin{minipage}[t]{\widthof{\includegraphics{#2}}}%
    {\fontsize{8bp}{9bp}\selectfont\textbf{#1}}\par\vspace{1pt}%
    \includegraphics{#2}%
  \end{minipage}%
}
\newcommand{\panelrow}[1]{\noindent#1\par\vspace{3mm}}
