%% paper/notes/notes-preamble.tex
%% Shared preamble for every lab-notes document in paper/notes/.
%%
%% These are NOTES, not the manuscript -- but they deliberately wear the manuscript's
%% EDITING look (paper/nature-biotech/editing.tex), so a figure drafted here appears at
%% the same size and type it will have in the paper, and prose reads at the same measure.
%% Everything that makes that look is inherited rather than re-declared:
%%
%%   sn-jnl.cls, sn-nature.bst, references.bib  -- symlinks to ../nature-biotech/
%%   preamble.tex                               -- \input directly from ../nature-biotech/
%%
%% so a change to the manuscript's packages or house macros reaches these notes with no
%% second edit. What differs from editing.tex is only what a note needs and a journal
%% submission does not: a table of contents, section numbering, and hyperlinks.

%% The manuscript's shared preamble: graphicx, \tcfig, amsmath, the proof house style,
%% booktabs, xcolor, pifont, placeins, listings, ... (see ../nature-biotech/preamble.tex).
\input{../nature-biotech/preamble}

%% ---- Same page geometry as editing.tex -------------------------------------------
%% A4 with print-approximate margins: a 182 mm text block (210 - 2*14) so a figure drawn
%% to Nature's 180 mm full width sits flush at 1:1 and a 88 mm column figure is ~0.49\textwidth.
%% twoside=false so margins do not alternate -- these are read on screen, not bound.
\geometry{a4paper,left=14mm,right=14mm,top=15mm,bottom=22mm,twoside=false,%
  bindingoffset=0mm,marginparwidth=0pt,marginparsep=0pt}

%% ---- Same block-paragraph drafting style as editing.tex --------------------------
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5\baselineskip plus 0.1\baselineskip}

%% ---- Notes-only: no-op the manuscript's editing-view macros ----------------------
%% The house macros \cb (char budget), \wc (word count) and \secstatus (stoplight chip)
%% are defined in editing.tex, not in the shared preamble, so a note that borrows a
%% paragraph from the manuscript would fail to compile without them. Define them as
%% no-ops -- a note has no word budget and no submission-readiness state.
\providecommand{\cb}[1]{}
\providecommand{\wc}[1]{}
\providecommand{\secstatus}[1]{}

%% ---- Notes-only: contents, numbering, links --------------------------------------
\definecolor{tclink}{HTML}{6C8EBF}% palette blue -- the only non-black text colour used
%% NB: sn-jnl.cls ALREADY loads hyperref, so \usepackage[...]{hyperref} here is an option
%% clash. Configure it with \hypersetup instead.
%%
%% hypertexnames=false: the class drives float captions through its own machinery, which emits
%% a named destination per figure; hyperref's counter-derived names then collide with those and
%% xdvipdfmx warns "Object @figure.N already defined" for every figure. Letting hyperref number
%% its destinations sequentially instead of by counter removes the collision -- \ref and the
%% bookmark tree are unaffected.
\hypersetup{hypertexnames=false,
  colorlinks=true,linkcolor=tclink,citecolor=tclink,urlcolor=tclink,
  bookmarksnumbered=true,bookmarksopen=true}
\usepackage{bookmark}

%% Number sections (the manuscript's Results/Methods are numbered; notes want the same,
%% plus subsubsections, because a note goes deeper than a paper section does) and show
%% three levels in the contents.
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{2}

%% ---- \incsvg -- include a matplotlib SVG as VECTOR --------------------------------
%% The repo figure standard has plotting scripts write BOTH .png and .svg to
%% ASSET_IMAGES_DIR, and notes embed the .svg (vector, and the same file that goes into
%% draw.io). LaTeX cannot read SVG, so the Makefile converts every \incsvg{...} target
%% to PDF with rsvg-convert into build/figs/ (flattening the path to a single basename),
%% and this macro includes that PDF. Photos and plate overlays are PNG and use plain
%% \includegraphics{assets/...} instead -- they are rasters, nothing to preserve.
%%
%%   \incsvg{W019-echo-crispr-array/run3/run3_se_batch_effect}
%%
%% Path is relative to assets/ (the symlink to notes/assets/images), WITHOUT the extension.
%%
%% NO DEFAULT WIDTH -- the figure is placed at its TRUE SIZE, exactly as the manuscript's
%% \tcfig does and for the same reason: these panels are authored at a real print width
%% (PANEL_WIDTHS_MM) with 6 pt type, so scaling them to \linewidth would silently resize the
%% type and defeat the WYSIWYG contract the editing view exists to provide. Pass an explicit
%% [width=...] ONLY to deliberately shrink an oversized draft panel, and treat that as a sign
%% the panel should be re-authored at a standard width instead.
\newcommand{\incsvg}[2][]{%
  \includegraphics[#1]{build/figs/\detokenize{#2}.pdf}%
}

%% ---- Float placement ------------------------------------------------------------
%% A note is read top-to-bottom as an argument, so a figure drifting two pages from the
%% sentence that introduces it is worse than a slightly loose page. Let floats take much
%% more of a page before LaTeX defers them, and cut the stretch it inserts around them --
%% the defaults are tuned for a journal article with few, small figures and leave big
%% vertical holes in a figure-dense note.
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.85}
\renewcommand{\textfraction}{0.08}
\renewcommand{\floatpagefraction}{0.75}
\setlength{\floatsep}{8pt plus 2pt minus 2pt}
\setlength{\textfloatsep}{10pt plus 2pt minus 3pt}
\setlength{\intextsep}{8pt plus 2pt minus 2pt}

%% Figure + caption for a note. Mirrors the manuscript's \tcfig -- the graphic is measured
%% into a save-box and the caption set to the IMAGE's width, so a half-column panel does not
%% get a full-width caption running under it.
%%
%% With ONE departure: a caption is never narrower than \notefigmincap. A panel authored at
%% the 88.5 mm half_plus width would otherwise carry a caption less than half the text
%% measure, which sets as a thin ragged column and reads badly. Below that threshold the
%% caption widens to the minimum and the pair is centred together.
\newsavebox{\notefigbox}
\newlength{\notefigcapwd}
\newlength{\notefigmincap}
\setlength{\notefigmincap}{0.62\textwidth}
\newcommand{\notefig}[2]{%
  \begin{figure}[htbp]\centering
  \sbox{\notefigbox}{#1}%
  \setlength{\notefigcapwd}{\wd\notefigbox}%
  \ifdim\notefigcapwd<\notefigmincap \setlength{\notefigcapwd}{\notefigmincap}\fi
  \usebox{\notefigbox}\par
  \begin{minipage}{\notefigcapwd}%
  \caption{#2}%
  \end{minipage}%
  \end{figure}%
}
