% ---------------------------------------------------------------------------
% CANONICAL NOTE STYLE, matched to paper/nature-biotech/editing.pdf.
% The paper's drafting view is the house look for notes-tex, so the heading shapes,
% paragraph handling and contents styling below are copied from sn-jnl.cls +
% editing.tex rather than invented. See bib_tex_pdf.sh for the page geometry half.
% ---------------------------------------------------------------------------

% Section headings. sn-jnl sets \section bold roman at 12bp and \subsection bold at
% 10bp with tight leading; \subsubsection is a bold run-in. Reproduced with titlesec so
% a pandoc note and a paper section have the same silhouette.
\usepackage{titlesec}
\titleformat{\section}{\normalfont\fontsize{12bp}{14bp}\bfseries}{\thesection}{1em}{}
\titleformat{\subsection}{\normalfont\fontsize{10bp}{12bp}\bfseries}{\thesubsection}{1em}{}
\titleformat{\subsubsection}[runin]%
  {\normalfont\fontsize{10bp}{12bp}\bfseries}{\thesubsubsection}{0.5em}{}[.\enspace]
\titlespacing*{\section}{0pt}{1.6ex plus .2ex}{0.8ex}
\titlespacing*{\subsection}{0pt}{1.4ex plus .2ex}{0.6ex}
\titlespacing*{\subsubsection}{0pt}{1.2ex plus .2ex}{0.5em}

% Block paragraphs: no first-line indent, a real gap between paragraphs. This is
% editing.tex's drafting choice, and it is what makes a dense note readable.
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5\baselineskip plus 0.1\baselineskip}

% Contents page, styled as editing.tex styles the paper's. Links are BLACK here: a
% contents page rendered in link-blue becomes one blue block and nothing stands out.
% Scoped so citations and cross-references keep their colours in the body.
% tocloft is GUARDED because it is not in every TeX installation, and notes are built
% on more than one machine. Without it the contents page still renders, just with
% LaTeX's default spacing; with it, it matches the paper. A missing optional package
% must never take down a note build.
\IfFileExists{tocloft.sty}{%
  \usepackage{tocloft}%
  \setlength{\cftbeforesecskip}{1.2pt}%
  \setlength{\cftbeforesubsecskip}{0.6pt}%
  \renewcommand{\cftsecfont}{\normalsize\bfseries}%
  \renewcommand{\cftsecpagefont}{\normalsize\bfseries}%
  \renewcommand{\cftsubsecfont}{\small}%
  \renewcommand{\cftsubsecpagefont}{\small}%
}{}

% Title positioning
\usepackage{titling}
\setlength{\droptitle}{-4em} % Move title up
\posttitle{\par\end{center}\vspace{-4.5em}} % Distance between title and text

% Image quality improvements
\usepackage{graphicx}
\DeclareGraphicsExtensions{.png,.pdf,.jpg}

% Figures are placed WHERE THEY ARE WRITTEN, not floated. Two reasons, and the second
% one is a correctness bug rather than a preference. First, a note is read in source
% order and a figure that drifts three pages from the sentence that introduces it stops
% being an illustration. Second, longtable and LaTeX's float mechanism do not cooperate:
% a float deferred onto a page that also starts a longtable makes longtable compute the
% page height before the float is added, and the page then overflows. That failure is
% SILENT and it LOSES CONTENT -- it dropped Table 19 of the 026 flux-layer note, caption
% and body, off the bottom of page 13, and the build still exited 0. Forcing H placement
% removes every float from the document and with it the collision. Loaded before caption
% so the caption package sees the restyled float.
\usepackage{float}
\floatplacement{figure}{H}

% Captions, matched to the paper's figure treatment: a bold run-in label followed by
% justified detail, set SMALLER than the body so the caption reads as apparatus rather
% than as text. labelformat=empty because the caption text already carries its own
% "Figure N." / "Table N." -- pandoc has no cross-reference filter here, so the numbers
% are written by hand and a second automatic "Figure 1:" in front of them would read
% "Figure 1: Figure 1.".
\usepackage{caption}
\captionsetup{
  labelformat=empty,
  font=footnotesize,
  justification=justified,
  singlelinecheck=false,
  skip=6pt
}

% Math strikethrough for cancelled terms
\usepackage{cancel}

% Default image settings
\setkeys{Gin}{draft=false}
% NO forced width. The repo's figure standard is TRUE SIZE: a panel designed at 88.5 mm
% carries 6 pt type that prints at 6 pt only if it is placed at 88.5 mm. Forcing
% width=\linewidth stretches every figure to the 182 mm block and silently rescales that
% type. Pandoc's own template already caps images at the text block, which is the safety
% net; natural size is the default.

\usepackage{xcolor}
\definecolor{linkred}{RGB}{205,92,92}      % Indian Red for citations and internal links
\definecolor{urlblue}{RGB}{100,149,237}    % Cornflower Blue for URLs

\usepackage{hyperref}
% linkcolor is BLACK on purpose, matching editing.tex. `linkcolor` drives the contents
% page as well as internal cross-references, and a coloured contents page renders as one
% solid block of colour in which nothing stands out. Citations and external URLs keep
% their colours, where colour actually carries meaning.
\hypersetup{
  colorlinks=true,
  linkcolor=black,     % internal links + the contents page
  citecolor=linkred,   % citations
  urlcolor=urlblue,    % external URLs
  pdfborder={0 0 0}
}

% Table sizing. Pandoc emits longtable and sets column widths from the SOURCE markdown's
% character counts, so a table whose cells hold long paths or commands is laid out to the
% width of the widest literal and then overruns the text block. Shrinking the type inside
% the table is the fix that does not require restructuring every table: \footnotesize buys
% roughly 20 % more characters per line, and \setlength{\tabcolsep} recovers more by
% tightening the inter-column padding. Applied to longtable so it hits every pandoc table
% without touching inline math or code blocks.
% \makeatletter wraps the WHOLE conditional, not its body: \IfFileExists reads its
% branches as arguments, so every token inside is catcoded at that moment. Switching @ on
% inside the branch is too late and turns \z@ into \z followed by a printed "@", which
% surfaces as "Missing \begin{document}".
\makeatletter
\IfFileExists{etoolbox.sty}{%
  \usepackage{etoolbox}%
  \AtBeginEnvironment{longtable}{\footnotesize\setlength{\tabcolsep}{4pt}}%
  \AtBeginEnvironment{tabular}{\footnotesize\setlength{\tabcolsep}{4pt}}%
  % A table and its caption belong on ONE page. Pandoc emits every table as a longtable,
  % and longtable's default inter-row breakpoint is free, so a six-row table sitting near
  % a page bottom splits rather than moving down whole: TeX scores an underfull page as
  % infinitely bad, so filling the page always beats leaving a gap, and a mere penalty of
  % 9999 loses that comparison. Only \nobreak actually moves the table.
  %
  % \nobreak is also how a table gets SILENTLY TRUNCATED, so it is switched on per table
  % rather than always. An unbreakable run longer than a page overflows the page and the
  % overflow is dropped, with the build still exiting 0 -- exactly the failure this block
  % exists to prevent. The switch is thrown by assets/publish/filters/keep-tables.lua,
  % which sees the whole table and turns it on only for one small enough to fit. Default
  % OFF, so a table nobody measured behaves as plain longtable always did.
  % A plain \def flag rather than \newif: \newif inside the \IfFileExists branch leaves
  % that conditional unbalanced and the build dies with "Extra \else".
  \newcommand\tcltkeepon{\global\def\tcltkeepflag{1}}%
  \newcommand\tcltkeepoff{\global\def\tcltkeepflag{0}}%
  \tcltkeepoff%
  \newcommand\tcltrowbreak{\if1\tcltkeepflag\nobreak\else\penalty9999 \fi}%
  \patchcmd{\LT@tabularcr}%
    {{\LT@t@bularcr}}%
    {{\def\crcr{\LT@crcr\noalign{\tcltrowbreak}}\let\cr\crcr\LT@t@bularcr}}%
    {}{\PackageWarning{header-includes}{longtable row-break patch did not apply}}%
}{}
\makeatother
